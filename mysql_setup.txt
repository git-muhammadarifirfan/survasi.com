PRASYARAT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ✓ MySQL 8.0+ atau MariaDB 10.5+ sudah terinstall
  ✓ Node.js 18+ sudah terinstall
  ✓ Akses ke MySQL sebagai root (atau user dengan hak CREATE DATABASE)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
LANGKAH 1 — BUAT DATABASE USER (Opsional, direkomendasikan)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Login ke MySQL sebagai root:
    mysql -u root -p

  Jalankan perintah berikut (ganti 'password_aman' dengan password yang kuat):
  ┌──────────────────────────────────────────────────────────────────────────┐
  │  CREATE USER 'bsan_user'@'localhost' IDENTIFIED BY 'password_aman';     │
  │  GRANT ALL PRIVILEGES ON db_survasi.* TO 'bsan_user'@'%';   │
  │  FLUSH PRIVILEGES;                                                       │
  │  EXIT;                                                                   │
  └──────────────────────────────────────────────────────────────────────────┘

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
LANGKAH 2 — INISIALISASI SCHEMA DATABASE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Jalankan dari root folder project (F:\1.PROJECT\survasi.com):
  ┌──────────────────────────────────────────────────────────────────────────┐
  │  mysql -u root -p < database/schema.sql                                 │
  └──────────────────────────────────────────────────────────────────────────┘

  Ini akan membuat database 'db_survasi' dengan:
    - 20 tabel utama (users, satuan_pendidikan, responden_survey, dll)
    - 3 views (v_kecamatan_stats, v_kabupaten_stats, v_sel_sesi_scores)
    - Trigger untuk update otomatis
    - Index dan Foreign Key constraints

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
LANGKAH 3 — IMPORT DATA AWAL (SEED)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Jalankan berurutan:
  ┌──────────────────────────────────────────────────────────────────────────┐
  │  mysql -u root -p db_survasi < database/seed.sql             │
  │  mysql -u root -p db_survasi < database/seed-sel-indikator.sql │
  │  mysql -u root -p db_survasi < database/seed-pertanyaan.sql  │
  └──────────────────────────────────────────────────────────────────────────┘

  seed.sql berisi:
    - Data provinsi, kabupaten (Sidoarjo, Batu, Tuban), kecamatan
    - 3 modul BSAN & 5 dimensi SEL
    - 2 akun default (admin + pengawas demo)

LANGKAH 5 — SETUP AKUN PENGAWAS (per Sekolah)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Setiap sekolah mendapat 1 akun pengawas dengan:
    Login    : NPSN sekolah
    Password : kata_pertama_nama_sekolah_lowercase + 4_digit_terakhir_NPSN

  Contoh Password:
    Nama: "SDN Candi 1"    NPSN: 20512345  →  password: sdn2345
    Nama: "MI Al-Hidayah"  NPSN: 60712100  →  password: mi2100
    Nama: "SMP Negeri 1"   NPSN: 20502100  →  password: smp2100
    Nama: "SDN Waru 2"     NPSN: 20510002  →  password: sdn0002

  Install dependencies backend dulu (jika belum):
  ┌──────────────────────────────────────────────────────────────────────────┐
  │  cd server                                                               │
  │  npm install                                                             │
  │  cd ..                                                                   │
  └──────────────────────────────────────────────────────────────────────────┘

  ┌──────────────────────────────────────────────────────────────────────────┐
  │  copy server\.env.example server\.env                                   │
  │  (edit server\.env sesuai konfigurasi MySQL kamu)                       │
  └──────────────────────────────────────────────────────────────────────────┘

  Jalankan script generator:
  ┌──────────────────────────────────────────────────────────────────────────┐
  │  node database/generate-pengawas-accounts.js                            │
  └──────────────────────────────────────────────────────────────────────────┘

  Script akan auto-generate akun untuk semua sekolah yang belum punya akun.

API ENDPOINTS REFERENCE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    POST   /auth/login            → Login (email atau NPSN)

    PUT    /sekolah/:id               → Update (admin only)
    GET    /sekolah/kabupaten         → Dropdown kabupaten
    GET    /sekolah/kecamatan         → Dropdown kecamatan

  Responden:
    GET    /responden                 → List responden
    GET    /responden/summary         → KPI responden
    GET    /responden/distribusi      → Distribusi posisi
    GET    /responden/export          → Export semua data

  Analisis:
    GET    /analisis/modul-progress   → Ring chart modul BSAN
    GET    /analisis/modul-detail     → Breakdown dimensi SEL
    GET    /analisis/proporsi         → Proporsi & distribusi modul
    GET    /analisis/funnel           → Gap funnel 5 tahap
    GET    /analisis/matriks          → Matriks 4 kuadran
    GET    /analisis/tantangan        → Top tantangan implementasi

  SEL:
    GET    /sel/indikator             → Master indikator SEL

    PUT    /sel/indikator/:id         → Edit indikator (admin)
    DELETE /sel/indikator/:id         → Hapus indikator (admin)
    POST   /sel/sesi                  → Submit sesi observasi
    GET    /sel/sesi                  → List sesi observasi
    GET    /sel/analisis/heatmap      → Heatmap skor kecamatan
    GET    /sel/analisis/radar/:id    → Radar benchmarking sekolah
    GET    /sel/analisis/summary      → KPI SEL
    GET    /sel/analisis/matriks      → Matriks SEL vs kuisioner

  Suara Responden:
    GET    /suara                     → List komentar

    GET    /setting/profile           → Profil user

    PUT    /setting/preferences       → Update preferensi
    GET    /setting/notifikasi        → List notifikasi

    GET    /setting/activity-log      → Log aktivitas

TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Error "Access denied for user":
    → Periksa DB_USER dan DB_PASS di server/.env

  Error "Unknown database db_survasi":
    → Jalankan dulu langkah 2 (schema.sql)

  Error "Table doesn't exist":
    → Jalankan semua file seed secara berurutan

  Error CORS di browser:
    → Pastikan CORS_ORIGINS di server/.env mencantumkan URL frontend

  Login gagal "Email/NPSN atau password salah":
    → Pastikan seed.sql sudah dijalankan
    → Coba login dengan: admin@survasi.com / admin

  Script generate-pengawas gagal "bcryptjs not found":
    → cd server && npm install && cd ..
    → node database/generate-pengawas-accounts.js
