Panduan ini berisi langkah-langkah lengkap membuat dan mengisi database MySQL 'db_survasi'
mulai dari NOL (Fresh Install / Database Kosong) hingga backend & frontend dapat berjalan.

Ikuti petunjuk di bawah secara BERURUTAN!

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
LANGKAH 0 — PRASYARAT APLIKASI & TERMINAL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Pastikan Service MySQL sudah berjalan di komputer kamu (XAMPP / MySQL Server / MariaDB).
2. Pastikan Node.js (v18 atau lebih baru) sudah terinstall.
3. Buka Terminal / Command Prompt (cmd) / PowerShell di folder utama project:
   `F:\1.PROJECT\survasi.com`


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
LANGKAH 1 — BUAT DATABASE & USER MYSQL (Via MySQL CLI)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Buka Terminal/CMD, lalu masuk ke prompt MySQL sebagai root:
  mysql -u root -p

(Tekan Enter, lalu masukkan password root MySQL kamu).

Di dalam prompt `mysql>`, jalankan perintah SQL berikut satu per satu:

  -- 1. Buat Database Baru 'db_survasi'
  CREATE DATABASE IF NOT EXISTS db_survasi
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

  -- 2. (Opsional) Buat User khusus untuk aplikasi
  CREATE USER IF NOT EXISTS 'bsan_user'@'localhost' IDENTIFIED BY 'password_aman';
  GRANT ALL PRIVILEGES ON db_survasi.* TO 'bsan_user'@'%';
  FLUSH PRIVILEGES;

  -- 3. Keluar dari MySQL CLI
  EXIT;


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
LANGKAH 2 — IMPORT STRUCTURE (SCHEMA DDL)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Jalankan perintah ini di Terminal / CMD (dari folder root project `F:\1.PROJECT\survasi.com`):

  mysql -u root -p db_survasi < database/schema.sql

(Sistem akan membuat 20 tabel utama, 3 views, trigger, dan foreign keys secara otomatis).


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
LANGKAH 3 — IMPORT DATA REFERENSI (SEED DATA)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Jalankan perintah berikut secara BERURUTAN:

  -- 1. Import Data Wilayah, Modul BSAN, Dimensi SEL, & User Admin Default
  mysql -u root -p db_survasi < database/seed.sql

  -- 2. Import 55 Indikator Observasi SEL (Guru & Murid)
  mysql -u root -p db_survasi < database/seed-sel-indikator.sql

  -- 3. Import 37 Pertanyaan Kuisioner Survey BSAN
  mysql -u root -p db_survasi < database/seed-pertanyaan.sql


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
LANGKAH 4 — IMPORT MASTER DATA SEKOLAH & RESPONDEN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Import data master sekolah (200+ SD/MI) dan data sampel responden:

  -- 1. Import Master Data Sekolah
  mysql -u root -p db_survasi < database/import-sekolah.sql

  -- 2. Import Data Responden Survey
  mysql -u root -p db_survasi < database/import-responden.sql


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
LANGKAH 5 — AUTO-GENERATE AKUN PENGAWAS SEKOLAH (NPSN)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Setiap sekolah di database akan otomatis dibuatkan akun pengawas login berbasis NPSN.

1. Masuk ke folder backend server dan install dependency:
   cd server
   npm install
   cd ..

2. Salin file konfigurasi .env backend (jika belum ada):
   copy server\.env.example server\.env

3. Buka file `server\.env` dengan Text Editor/VS Code dan sesuaikan password MySQL kamu:
   DB_HOST=localhost
   DB_PORT=3306
   DB_USER=root           <-- Sesuaikan user MySQL kamu
   DB_PASS=password_kamu  <-- Sesuaikan password MySQL kamu
   DB_NAME=db_survasi

4. Jalankan script generator akun pengawas dari root project:
   node database/generate-pengawas-accounts.js

   (Script akan memproses semua sekolah dan menggenerate password hash Bcrypt).

LANGKAH 6 — VERIFIKASI HASIL IMPORT DATABASE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Jalankan script verifikasi untuk memastikan semua tabel & data terisi dengan sempurna:

  mysql -u root -p db_survasi < database/verify.sql

Pastikan seluruh indikator pengujian pada output Terminal mengembalikan status `✓ PASS`.

LANGKAH 7 — JALANKAN SERVER BACKEND & FRONTEND
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Jalankan Backend Server API (Terminal 1):
   cd server
   npm run dev

   Server akan berjalan di: http://localhost:3001
   Health Check Status   : http://localhost:3001/health

2. Jalankan Frontend Web Application (Terminal 2 - New Window):
   npm run dev

   Buka Browser di: http://localhost:5173

SOLUSI TROUBLESHOOTING & ERROR POPULER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Error: "Access denied for user 'root'@'localhost'"
   ➔ Password MySQL root kamu salah. Sesuaikan `-p` dengan password yang benar.

2. Error: "Unknown database 'db_survasi'"
   ➔ Kamu belum membuat database di Langkah 1 (`CREATE DATABASE db_survasi;`).

3. Error: "Table 'db_survasi.users' doesn't exist"
   ➔ Langkah 2 (`schema.sql`) belum dijalankan atau gagal.

4. Error: "bcryptjs not found" saat running generator
   ➔ Jalankan command `cd server && npm install` terlebih dahulu.
